from ultralytics import YOLO
import os

yaml_path = f"{dataset.location}/data.yaml"
print(f"✅ เจอไฟล์ Dataset ของคุณแล้วที่: {yaml_path}")

# ตรวจขนาด dataset ก่อน
train_count = len(os.listdir(f"{dataset.location}/train/images"))
valid_count = len(os.listdir(f"{dataset.location}/valid/images"))
print(f"📊 Train: {train_count} | Valid: {valid_count}")

model = YOLO('yolo11m.pt')  # ใช้ nano แทน small สำหรับ dataset เล็ก

results = model.train(
    data=yaml_path,
    epochs=100,          # ลดลง — ใช้ early stopping แทน
    patience=20,         # หยุดเองถ้า val loss ไม่ดีขึ้น 20 epochs
    imgsz=1280,           # ลดลงจาก 1280 → gradient เสถียรขึ้น
    batch=8,            # เพิ่มได้เพราะภาพเล็กลง
    optimizer='AdamW',
    lr0=0.001,
    lrf=0.01,

    # Augmentation สำหรับปัญหาเงา/แสงน้อย
    hsv_v=0.5,           # สุ่มความสว่าง (จำลองเงา)
    hsv_s=0.5,           # สุ่ม saturation
    degrees=15,          # หมุนเล็กน้อย
    flipud=0.1,          # พลิกแนวตั้งบางครั้ง
    mosaic=1.0,          # รวม 4 ภาพ → เพิ่ม context หลากหลาย
    mixup=0.1,           # blend สองภาพ → ลด overfit

    project='/content/drive/MyDrive/Petanque_YOLO',
    name='train_v4',
    save_period=10,      # save checkpoint ทุก 10 epochs ป้องกันเน็ตหลุด
)
